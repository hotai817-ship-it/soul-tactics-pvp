<script>

    const socket = io();

    let currentRoomId = null;


    function playSound() {

        try {

            const ctx = new (window.AudioContext || window.webkitAudioContext)();

            const osc = ctx.createOscillator();

            const gain = ctx.createGain();

            osc.connect(gain); gain.connect(ctx.destination);

            osc.frequency.setValueAtTime(600, ctx.currentTime);

            gain.gain.setValueAtTime(0.1, ctx.currentTime);

            osc.start(); osc.stop(ctx.currentTime + 0.08);

        } catch(e) {}

    }


    function createRoom() {

        const roomId = document.getElementById('roomInput').value.trim();

        if (!roomId) { alert("請輸入房間號碼！"); return; }

        socket.emit('create_room', roomId);

    }


    function joinRoom() {

        const roomId = document.getElementById('roomInput').value.trim();

        if (!roomId) { alert("請輸入房間號碼！"); return; }

        socket.emit('join_room', roomId);

    }


    socket.on('room_created', (roomId) => {

        currentRoomId = roomId;

        document.getElementById('lobby').innerHTML = `<h3 style='color:var(--gold);'>🏠 房間 [${roomId}] 建立成功！</h3><p style='color:#38bdf8;'>等待對手加入中...</p>`;

    });


    socket.on('error_msg', (msg) => { alert(msg); });


    socket.on('game_start', ({ roomId }) => {

        currentRoomId = roomId;

        document.getElementById('lobby').classList.add('hidden');

        document.getElementById('gameArea').classList.remove('hidden');

        appendLog(`🎉 成功進入房間 [${roomId}]，請先部署歷史人物或使用輔助卡！`);

    });


    socket.on('game_state_update', ({ myState, opState, isMyTurn }) => {

        document.getElementById('myHp').innerText = myState.hp;

        document.getElementById('opHp').innerText = opState.hp;


        const indicator = document.getElementById('turnIndicator');

        if (isMyTurn) {

            indicator.innerText = "⭐ 您的回合 (可佈署卡牌或點擊英雄答題攻擊)";

            indicator.style.color = "#4ade80";

        } else {

            indicator.innerText = "⏳ 對手回合中...";

            indicator.style.color = "#f87171";

        }


        // 渲染手牌

        const handDiv = document.getElementById('myHand');

        handDiv.innerHTML = "";

        myState.hand.forEach((card, idx) => {

            let tagColor = card.type === '人物' ? '#78350f' : '#1e3a8a';

            handDiv.innerHTML += `

                <div class="card" onclick="playCard(${idx})">

                    <div style="font-size:0.75em; font-weight:bold; color:${tagColor};">[${card.dynasty}] ${card.type}</div>

                    <div style="font-size:0.95em; font-weight:bold; color:#78350f; text-align:center;">${card.name}</div>

                    <div style="font-size:0.65em; color:#475569; overflow:hidden; max-height:45px;">${card.skill}: ${card.desc}</div>

                    <div style="font-size:0.7em; background:#fef3c7; font-weight:bold; color:#b45309; text-align:center;">點擊部署</div>

                </div>`;

        });


        // 渲染我方主戰人物與支援區

        const myActive = document.getElementById('myActive');

        let supportHtml = myState.supportArea.map(c => `<span style="font-size:0.75rem; background:#1e3a8a; color:white; padding:4px 8px; border-radius:4px; margin-right:4px;">🛡️ ${c.name}</span>`).join('');

        

        if (myState.active) {

            myActive.innerHTML = `

                <div style="display:flex; flex-direction:column; align-items:center; gap:5px; width:100%;">

                    <div style="display:flex; gap:10px; align-items:center;">

                        <div class="card" style="border:3px solid #22c55e;" onclick="triggerAttack()">

                            <div style="font-size:0.75em; font-weight:bold; color:#78350f;">[${myState.active.dynasty}] 主戰人物</div>

                            <div style="font-size:0.95em; font-weight:bold; color:#78350f; text-align:center;">${myState.active.name}</div>

                            <div style="font-size:0.65em; color:#475569;">${myState.active.skill}</div>

                            <div style="font-size:0.7em; background:#dcfce7; color:#15803d; font-weight:bold; text-align:center;">⚔️ 點擊答題攻擊</div>

                        </div>

                    </div>

                    ${supportHtml ? `<div style="font-size:0.85rem; color:#38bdf8;">支援區: ${supportHtml}</div>` : ''}

                </div>`;

        } else {

            myActive.innerHTML = `<div style="color:#64748b; font-size:0.95rem;">請從手牌點擊歷史人物卡部署到戰陣上 ${supportHtml ? '<br>支援區: ' + supportHtml : ''}</div>`;

        }


        // 渲染對手陣地

        const opActive = document.getElementById('opActive');

        let opSupportHtml = opState.supportArea.map(c => `<span style="font-size:0.75rem; background:#1e3a8a; color:white; padding:4px 8px; border-radius:4px;">🛡️ ${c.name}</span>`).join('');

        if (opState.active) {

            opActive.innerHTML = `

                <div style="display:flex; flex-direction:column; align-items:center; gap:5px;">

                    <div class="card" style="border:3px solid #ef4444;">

                        <div style="font-size:0.75em; font-weight:bold; color:#78350f;">[${opState.active.dynasty}] 對手主戰</div>

                        <div style="font-size:0.95em; font-weight:bold; color:#78350f; text-align:center;">${opState.active.name}</div>

                        <div style="font-size:0.65em; color:#475569;">${opState.active.skill}</div>

                        <div style="font-size:0.7em; background:#fee2e2; color:#991b1b; font-weight:bold; text-align:center;">部署中</div>

                    </div>

                    ${opSupportHtml ? `<div style="font-size:0.85rem; color:#38bdf8;">支援區: ${opSupportHtml}</div>` : ''}

                </div>`;

        } else {

            opActive.innerHTML = `<div style="color:#64748b; font-size:0.95rem;">對手尚未部署歷史人物</div>`;

        }

    });


    function playCard(cardIndex) {

        socket.emit('play_card', { roomId: currentRoomId, cardIndex });

    }


    // 點擊主戰人物觸發歷史考驗題問

    function triggerAttack() {

        const questions = [

            { q: "秦王政自認為「XX三皇，XX五帝」，於是把三「皇」五「帝」名稱合稱為「皇帝」。請填空：", ans: "功過三皇，德兼五帝", opts: ["A. 功過三皇，德兼五帝", "B. 德高三皇，功過五帝", "C. 德兼三皇，功過五帝"] },

            { q: "漢武帝採納主父偃的建議，頒行哪一項法令削弱諸侯王勢力？", ans: "推恩令", opts: ["A. 削藩令", "B. 推恩令", "C. 汉律令"] },

            { q: "唐太宗在位期間政治清明，史稱為什麼？", ans: "貞觀之治", opts: ["A. 開元之治", "B. 貞觀之治", "C. 文景之治"] }

        ];

        let q = questions[Math.floor(Math.random() * questions.length)];

        document.getElementById('qText').innerText = q.q;

        const optDiv = document.getElementById('qOptions');

        optDiv.innerHTML = "";

        q.opts.forEach(opt => {

            let isCorrect = opt.includes(q.ans) || opt.charAt(0) === 'A';

            optDiv.innerHTML += `<button class="modal-btn" onclick="submitAnswer(${isCorrect})">${opt}</button>`;

        });

        document.getElementById('questionModal').classList.remove('hidden');

    }


    function submitAnswer(isCorrect) {

        document.getElementById('questionModal').classList.add('hidden');

        socket.emit('attack', { roomId: currentRoomId, answerCorrect: isCorrect });

    }


    function endTurn() { 

        socket.emit('end_turn', currentRoomId); 

    }


    socket.on('battle_log', (data) => { appendLog(data.msg); });

    socket.on('game_over', ({ winner }) => {

        if (winner === socket.id) {

            alert("🎉 恭喜您在《英魂決》對戰中獲得勝利！");

        } else {

            alert("❌ 勝敗乃兵家常事，少俠請重新來過！");

        }

        location.reload();

    });


    function appendLog(text) {

        const box = document.getElementById('logBox');

        box.innerHTML += `<div>> ${text}</div>`;

        box.scrollTop = box.scrollHeight;

    }

</script>
